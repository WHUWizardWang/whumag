Datapoint ReadData::setDataResult(Datapoint& datapoint, double interval)
{
    Datapoint dataresult;
    double xMax = numeric_limits<double>::min();
    double xMin = numeric_limits<double>::max();
    double yMax = numeric_limits<double>::min();
    double yMin = numeric_limits<double>::max();
    for (auto& elem : datapoint)
    {
        double x, y;
        x = elem.second.X*10;
        y = elem.second.Y*10;
        elem.second.X = round(x)/10.0;
        elem.second.Y = round(y)/10.0;
        xMax = max(xMax, elem.second.X);
        xMin = min(xMin, elem.second.X);
        yMax = max(yMax, elem.second.Y);
        yMin = min(yMin, elem.second.Y);
    }
    for (double x = xMin; x <= xMax; x += interval)
    {
        for (double y = yMin; y <= yMax; y += interval)
        {
            SinglePoint point;
            point.X = x;
            point.lat = x;
            point.Y = y;
            point.lon = y;
            point.tMagnetic = 0;
            dataresult.insert(make_pair(dataresult.size(), point));
        }
    }
    return dataresult;
}
